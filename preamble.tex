% Signature block macro: left-aligns the name (start at left of block)
% and centers the role in parentheses directly under the name text.
\newlength{\signnamewidth}
\newcommand{\signatureblock}[2]{%
  \noindent
  \begin{minipage}[t]{0.6\textwidth}
    \raggedright
    % measure the name width and typeset a parbox of that width so the
    % role can be centered beneath it
    \settowidth{\signnamewidth}{#1}%%
    \parbox{\signnamewidth}{#1}\\[-0.5ex]
    {\small\makebox[\signnamewidth][c]{(#2)}}%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.35\textwidth}
    \raggedleft
    Date: \rule{4cm}{0.4pt}
  \end{minipage}
}
